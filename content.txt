import Card from "./card";
import { Suit } from "./types";

class Deck {
  private deck: Card[] = [];

  constructor() {
    this.reset();
  }

  reset(): void {
    this.deck = [];

    const suits: Suit[] = ["♠", "♥", "♦", "♣"];

    for (const suit of suits) {
      for (let value = 1; value <= 13; value++) {
        this.deck.push(new Card(value, suit));
      }
    }

    this.shuffle();
  }

  private shuffle(): void {
    for (let i = this.deck.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));

      [this.deck[i], this.deck[j]] = [
        this.deck[j],
        this.deck[i],
      ];
    }
  }

  deal(count: number): Card[] {
    const cards: Card[] = [];

    for (let i = 0; i < count; i++) {
      const card = this.deck.shift();

      if (card) {
        cards.push(card);
      }
    }

    return cards;
  }
}

export default Deck;

import { ICard } from "../game/types";

export function getHandValue(cards: ICard[]): number {
  let total = 0;
  let aces = 0;

  for (const card of cards) {
    if (card.value === 1) {
      total += 11;
      aces++;
    } else if (card.value >= 11) {
      total += 10;
    } else {
      total += card.value;
    }
  }

  while (total > 21 && aces > 0) {
    total -= 10;
    aces--;
  }

  return total;
}

export function isBlackjack(cards: ICard[]): boolean {
  return cards.length === 2 && getHandValue(cards) === 21;
}

export function isBust(cards: ICard[]): boolean {
  return getHandValue(cards) > 21;
}

```ts
import Deck from "./deck";
import { GameStatus, ICard } from "./types";
import {
  getHandValue,
  isBlackjack,
  isBust,
} from "../utils/gameUtils";

class BlackjackGame {
  private deck: Deck;

  playerHand: ICard[] = [];
  dealerHand: ICard[] = [];

  balance: number;
  bet: number = 0;
  status: GameStatus = "idle";

  constructor(initialBalance: number = 1000) {
    this.deck = new Deck();
    this.balance = initialBalance;
  }

  startRound(bet: number): void {
    if (bet <= 0 || bet > this.balance) {
      return;
    }

    this.bet = bet;
    this.balance -= bet;

    this.deck.reset();

    this.playerHand = this.deck.deal(2);
    this.dealerHand = this.deck.deal(2);

    const playerBlackjack =
      isBlackjack(this.playerHand);

    const dealerBlackjack =
      isBlackjack(this.dealerHand);

    if (playerBlackjack && dealerBlackjack) {
      this.balance += bet;
      this.status = "push";
      return;
    }

    if (playerBlackjack) {
      this.balance += bet * 2.5;
      this.status = "blackjack";
      return;
    }

    if (dealerBlackjack) {
      this.status = "lost";
      return;
    }

    this.status = "playing";
  }

  hit(): void {
    if (this.status !== "playing") {
      return;
    }

    const newCards = this.deck.deal(1);

    this.playerHand = [
      ...this.playerHand,
      ...newCards,
    ];

    if (isBust(this.playerHand)) {
      this.status = "bust";
    }
  }

  stand(): void {
    if (this.status !== "playing") {
      return;
    }

    this.status = "dealer-turn";

    while (getHandValue(this.dealerHand) < 17) {
      const newCards = this.deck.deal(1);

      this.dealerHand = [
        ...this.dealerHand,
        ...newCards,
      ];
    }

    this.resolveRound();
  }

  private resolveRound(): void {
    const playerValue =
      getHandValue(this.playerHand);

    const dealerValue =
      getHandValue(this.dealerHand);

    if (dealerValue > 21) {
      this.balance += this.bet * 2;
      this.status = "won";
      return;
    }

    if (playerValue > dealerValue) {
      this.balance += this.bet * 2;
      this.status = "won";
      return;
    }

    if (playerValue === dealerValue) {
      this.balance += this.bet;
      this.status = "push";
      return;
    }

    this.status = "lost";
  }

  resetGame(): void {
    this.playerHand = [];
    this.dealerHand = [];
    this.bet = 0;
    this.status = "idle";
  }
}

export default BlackjackGame;

The player hit and dealer turn rules follow the logic from your original game.

    syncGame();

    syncGame();

    getHandValue(dealerHand);

  const isPlaying =
    status === "playing";

  const showBetting =
    status === "idle";

  return (
    <main className="game-table">
      <section className="dealer-section">
        <div className="section-title">
          DEALER
        </div>

        <Hand
          cards={dealerHand}
          hideSecondCard={isPlaying}
        />

        {dealerHand.length > 0 && (
          <div className="score">
            {isPlaying ? "?" : dealerValue}
          </div>
        )}
      </section>

      <div className="table-divider" />

      <section className="player-section">
        <div className="section-title">
          PLAYER
        </div>

        <Hand cards={playerHand} />

        {playerHand.length > 0 && (
          <div className="score">
            {playerValue}
          </div>
        )}
      </section>

      {showBetting && (
        <BettingPanel
          balance={balance}
          onBet={onBet}
        />
      )}

      {isPlaying && (
        <ActionButtons
          onHit={onHit}
          onStand={onStand}
        />
      )}

      {status !== "idle" &&
        status !== "playing" &&
        status !== "dealer-turn" && (
          <GameResult
            status={status}
            bet={bet}
            onNewRound={onNewRound}
          />
        )}
    </main>
  );
}

export default GameTable;
```
